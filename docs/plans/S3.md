## Review Focus

- KNP-03 starts above capacity: pulling stock out is fine; any inbound to KNP-03 is a CAPACITY violation even if it stays "less over" than before.

### Task 1: Validator

**Files:** Create `engine/rules.py`, Test `tests/test_rules.py`

**Interfaces:**
- Consumes: `Data` (raw `snapshot`, `inbound`, `stores`, `items`, `min_display`), `Features.sku_pos.vel_daily` (only for DONOR_COVER and IN_TRANSIT), C2 `groups`.
- Produces: `validate(data, feats, groups: list[dict]) -> dict`.

Checks per line (replay sorted by `ship_on`, then group order):
- `NON_POSITIVE_QTY` qty ≤ 0. `LEAD_TIME` `arrives_on − ship_on ≠ lead_days(from, to)` or `group.lead_days` mismatch.
- `FLAGSHIP_PROTECTED` giver is S++ and `(from, style_id)` in `protected_styles(data)`.
- `NEGATIVE_STOCK` giver store×SKU stock < 0 after the line. `WH_NEGATIVE` cumulative WH sends of a SKU up to `ship_on` > WH on hand + inbound with `expected_date ≤ ship_on`.
- `MIN_DISPLAY` giver store's cat_group total < tier min after the line.
- `CAPACITY` receiver store total > capacity after the line.
- `DONOR_COVER` STORE_TO_STORE only: giver style cover after (`Σ stock / Σ vel_daily` over style at giver) < 30.
- `IN_TRANSIT` WH_TO_STORE to a store×SKU with `in_transit > 0` whose `on_hand + in_transit ≥ ceil(vel_daily × (LEAD_WH[zone] + 14))` before the line, unless `problem_kind == "MIN_DISPLAY"`.
- `ROUND_TRIP` same store×SKU appears as both giver and receiver in the run.
Start stock per store×SKU = `on_hand` (already clipped) + `in_transit`.

import pytest
from engine.data import load_data
from engine.interfaces import build_features

@pytest.fixture(scope="module")
def df():

def grp(gid, typ, frm, to, sku, qty, ship="2026-11-01", arr=None, lead=None, kind="UNDERSTOCK"):
    from engine.data import lead_days
    d = load_data()
    lead = lead if lead is not None else lead_days(frm, to, d.stores)
    import pandas as pd
    arr = arr or (pd.Timestamp(ship) + pd.Timedelta(days=lead)).strftime("%Y-%m-%d")
    style = sku.rsplit("-", 1)[0]
    return {"group_id": gid, "type": typ, "problem_kind": kind, "style_id": style, "from": frm, "to": to,
            "ship_on": ship, "arrives_on": arr, "lead_days": lead, "qty": qty, "sizes": {sku.rsplit("-",1)[1]: qty},
            "lines": [{"sku_id": sku, "size": sku.rsplit("-", 1)[1], "qty": qty}]}

def rules_hit(df, groups):
    from engine.rules import validate
    return {v["rule"] for v in validate(*df, groups)["violations"]}

def test_empty_passes(df):

    assert validate(*df, []) == {"passed": True, "checked_moves": 0, "violations": []}

def test_flagship_protected(df):

def test_capacity_into_knp03(df):
    assert "CAPACITY" in rules_hit(df, [grp("G1", "WH_TO_STORE", "WH-KOL", "KNP-03", "KUR-09-XL", 1)])

def test_min_display_giver(df):
    assert "MIN_DISPLAY" in rules_hit(df, [grp("G1", "PULL_BACK", "MZF-01", "WH-KOL", "MOJ-01-9", 1)])   # MZF-01 holds 3 of MOJ-01-9; Mojari total 5 < min 8

def test_wh_negative_by_date(df):
    assert "WH_NEGATIVE" in rules_hit(df, [grp("G1", "WH_TO_STORE", "WH-KOL", "PAT-01", "KUR-01-XL", 100)])

def test_lead_time(df):
    assert "LEAD_TIME" in rules_hit(df, [grp("G1", "WH_TO_STORE", "WH-KOL", "PAT-01", "KUR-09-XL", 1, arr="2026-11-02", lead=1)])

def test_non_positive(df):
    assert "NON_POSITIVE_QTY" in rules_hit(df, [grp("G1", "WH_TO_STORE", "WH-KOL", "PAT-01", "KUR-09-XL", 0)])

def test_round_trip(df):
    g = [grp("G1", "WH_TO_STORE", "WH-KOL", "PAT-02", "KUR-09-XL", 1),
         grp("G2", "PULL_BACK", "PAT-02", "WH-KOL", "KUR-09-XL", 1)]

def test_clean_move_passes(df):
    assert rules_hit(df, [grp("G1", "WH_TO_STORE", "WH-KOL", "PAT-02", "KUR-09-XL", 2)]) <= {"IN_TRANSIT"}

### Task 2: Findings (flags)

**Files:** Create `engine/quality.py`, Test `tests/test_quality.py`

**Interfaces:** Produces `find_flags(data, feats) -> list[dict]`.

Rules:
- `DATA_QUALITY`: every `snapshot.on_hand_raw < 0` ("On hand is −4 … treated as 0, stock count needed"); one flag per store where units were sold in `stock_weekly` weeks with `days_in_stock == 0` (count units — stock records unreliable there).
- `REORDER`: style sold ≥ 50 units in the last 365 days, chain stock (stores + WH) ≤ 5% of that, and no inbound → "Sold out chain-wide (N sold in 12 months, M left, nothing inbound). Zero recent sales = sold out, not slow."
- `DEAD_STOCK`: style in stock (`days_in_stock == 7`) at ≥ 5 stores in all 12 weeks and ≤ 10 units sold chain-wide in the last 84 days.
- `NEW_STORE`: store opened ≤ 90 days before `as_of`.
- `CAPACITY`: store `utilisation > 1.0` ("520 units in a 250-unit store (208%)").

import pytest
from engine.run import run_engine

@pytest.fixture(scope="module")
def run(tmp_path_factory):
    return run_engine(trigger="test", strict=False, out_dir=tmp_path_factory.mktemp("out"))

def q(run, **kw):
    return sum(g["qty"] for g in run["groups"] if all(g[k] == v for k, v in kw.items()))

def test_validation_passes(run):                       # P0: no rule broken
    assert run["validation"]["passed"], run["validation"]["violations"][:5]

    assert q(run, **{"from": "KNP-03", "style_id": "SHW-02"}) >= 200

    assert q(run, to="MZF-01") >= 20

def test_p3_kur04_new_launch(run):

    assert len({g["to"] for g in run["groups"] if g["style_id"] == "KUR-01" and g["type"] == "WH_TO_STORE"}) >= 8
